<threat_model>
### T-03-01: Single-Shot Fault Injection Leakage and Indefinite Link Impairment
- **Threat:** Armed single-shot faults (`arm_drop_next_message`) or custom drop rates lingering across tests or unrelated transmissions, causing random message loss.
- **Impact:** Corruption of standard message flow, flaky integration test runs, and operator confusion.
- **Mitigation:**
  - In `SimulatedTransport.transmit()`, armed flags must auto-disarm immediately after one firing.
  - Implement explicit `disarm_all()` and `reset()` methods resetting all parameters to 0ms latency and 0% drops.
  - Add pytest teardown resetting transport state between all tests.

### T-03-02: Stale Worker Lease Starvation and Attempt Counter Overflow
- **Threat:** Crashed workers leave messages orphaned in `SENDING` status indefinitely. Furthermore, manual re-queuing of messages in `FAILED_FINAL` fails to reset `attempt_count`, immediately re-failing the message on its next lease.
- **Impact:** Messages become permanently stuck or instantly fail without attempting delivery.
- **Mitigation:**
  - Execute `RetryPolicy.recover_stranded_leases(session)` on application startup to sweep any dead in-flight leases.
  - In `MessagingService.retry_message()`, explicitly set `msg.attempt_count = 0` to grant a fresh quota of 5 delivery attempts.
  - In `RetryPolicy.reclaim_stale_leases()`, record `previous_worker_id` and `lease_expires_at` into `safe_metadata` to maintain tamper-evident audit trails.
</threat_model>

   Add method to `RetryPolicy`:
   ```python
   @classmethod
   def recover_stranded_leases(cls, session: Session) -> int:
       """Scans and transitions all orphaned SENDING messages left by previously terminated processes."""
       now = to_naive_utc(utcnow())
       stranded = (
           session.query(Message)
           .filter(Message.current_state == MessageState.SENDING.value)
           .all()
       )
       count = 0
       for msg in stranded:
           prev_worker_id = msg.worker_lease_id or "orphaned-worker"
           prev_lease_exp = msg.lease_expires_at.isoformat() if msg.lease_expires_at else None
           msg.current_state = MessageState.FAILED_RETRYABLE.value
           msg.next_attempt_at = now - timedelta(seconds=1)
           msg.worker_lease_id = None
           msg.lease_expires_at = None
           msg.last_error_code = "LEASE_EXPIRED"
           msg.last_error_at = now

           event = MessageEvent(
               event_id=generate_uuid(),
               message_id=msg.message_id,
               event_type="WORKER_LEASE_RECLAIMED",
               timestamp=now,
               source_endpoint="startup-supervisor",
               safe_metadata={
                   "previous_worker_id": prev_worker_id,
                   "lease_expires_at": prev_lease_exp,
                   "reason": "Stranded in-flight lease recovered upon service startup",
                   "startup_sweep": True,
               },
               error_code="LEASE_EXPIRED",
           )
           session.add(event)
           count += 1
       if count > 0:
           session.commit()
       return count
   ```

     ```typescript
     class AudioService {
       private ctx: AudioContext | null = null
       private isMuted: boolean = localStorage.getItem('shadowcomm_audio_muted') === 'true'

       private getContext(): AudioContext | null {
         if (typeof window === 'undefined') return null
         if (!this.ctx) {
           const AudioCtx = window.AudioContext || (window as any).webkitAudioContext
           if (AudioCtx) this.ctx = new AudioCtx()
         }
         if (this.ctx && this.ctx.state === 'suspended') {
           this.ctx.resume().catch(() => {})
         }
         return this.ctx
       }

       playSquelch() {
         if (this.isMuted) return
         const ctx = this.getContext()
         if (!ctx) return
         // Synthesize 60ms bandpass white noise burst
         const bufferSize = ctx.sampleRate * 0.06
         const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate)
         const output = buffer.getChannelData(0)
         for (let i = 0; i < bufferSize; i++) {
           output[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.5))
         }
         const whiteNoise = ctx.createBufferSource()
         whiteNoise.buffer = buffer

         const filter = ctx.createBiquadFilter()
         filter.type = 'bandpass'
         filter.frequency.value = 1200
         filter.Q.value = 3.0

         const gain = ctx.createGain()
         gain.gain.setValueAtTime(0.08, ctx.currentTime)
         gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.06)

         whiteNoise.connect(filter)
         filter.connect(gain)
         gain.connect(ctx.destination)
         whiteNoise.start()
       }

       playRestoreChime() {
         if (this.isMuted) return
         const ctx = this.getContext()
         if (!ctx) return
         // Synthesize pleasant 2-tone clean link chime (880Hz -> 1760Hz)
         const now = ctx.currentTime
         const osc = ctx.createOscillator()
         const gain = ctx.createGain()
         osc.type = 'sine'
         osc.frequency.setValueAtTime(880, now)
         osc.frequency.setValueAtTime(1760, now + 0.08)

         gain.gain.setValueAtTime(0.06, now)
         gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25)

         osc.connect(gain)
         gain.connect(ctx.destination)
         osc.start(now)
         osc.stop(now + 0.25)
       }

       toggleMute(): boolean {
         this.isMuted = !this.isMuted
         localStorage.setItem('shadowcomm_audio_muted', String(this.isMuted))
         return this.isMuted
       }

       getMuted(): boolean {
         return this.isMuted
       }
     }

     export const audioService = new AudioService()
     ```
